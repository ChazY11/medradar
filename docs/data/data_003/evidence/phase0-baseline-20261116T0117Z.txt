#!/usr/bin/env bash
# MedRadar health check (CLOUD-003). Liveness + readiness + resource checks.
# Exit code is always 0 so the timer stays quiet; status is in stdout/journal/state file.
set -u
APP="http://127.0.0.1:8787"
STATE="/var/lib/medradar-health"
mkdir -p "$STATE"
ts=$(date -Is)
crit=0; warn=0; msgs=""

add() { if [ -z "$msgs" ]; then msgs="$1"; else msgs="$msgs,$1"; fi; }

if systemctl is-active --quiet medradar-web; then live=up; else live=down; crit=1; add "systemd:down"; fi

code=$(curl -s -o /dev/null -m 5 -w '%{http_code}' "$APP/api/health" 2>/dev/null || echo 000)
if [ "$code" != "200" ]; then crit=1; live="${live}/http${code}"; add "http_health:$code"; fi

q=$(curl -s -m 10 "$APP/api/devices/quality" 2>/dev/null || true)
case "$q" in *index_build_ms*) ready=yes ;; *) ready=no; crit=1; add "device_index:not_ready" ;; esac

o=$(curl -s -m 10 "$APP/api/organizations" 2>/dev/null || true)
case "$o" in *medtronic*) org=yes ;; *) org=no; warn=1; add "org_query:fail" ;; esac

memavail=$(awk '/MemAvailable/{print int($2/1024)}' /proc/meminfo)
swapused=$(free -m | awk '/^Swap:/{print $3}')
diskavail=$(df -Pm / | awk 'NR==2{print $4}')
restarts=$(systemctl show medradar-web -p NRestarts --value 2>/dev/null || echo 0)

[ "${memavail:-0}" -lt 300 ] && { crit=1; add "mem_avail:${memavail}MB"; }
[ "${diskavail:-0}" -lt 5000 ] && { crit=1; add "disk_free:${diskavail}MB"; }
[ "${swapused:-0}" -gt 1024 ] && { warn=1; add "swap_used:${swapused}MB"; }
[ "${restarts:-0}" -gt 3 ] && { warn=1; add "nrestarts:${restarts}"; }

status=ok
[ "$warn" -eq 1 ] && status=warn
[ "$crit" -eq 1 ] && status=critical

printf '%s status=%s systemd=%s index=%s org_query=%s mem_avail=%sMB swap=%sMB disk_free=%sMB restarts=%s issues=%s\n' \
  "$ts" "$status" "$live" "$ready" "$org" "$memavail" "$swapused" "$diskavail" "$restarts" "${msgs:-none}"

printf '{"ts":"%s","status":"%s","systemd":"%s","device_index":"%s","org_query":"%s","mem_available_mb":%s,"swap_used_mb":%s,"disk_available_mb":%s,"nrestarts":%s,"issues":"%s"}\n' \
  "$ts" "$status" "$live" "$ready" "$org" "${memavail:-0}" "${swapused:-0}" "${diskavail:-0}" "${restarts:-0}" "${msgs:-none}" > "$STATE/last.json"

if [ "$status" != "ok" ]; then
  logger -t medradar-health -p user.warning "medradar health $status issues=${msgs:-none}"
fi
exit 0
