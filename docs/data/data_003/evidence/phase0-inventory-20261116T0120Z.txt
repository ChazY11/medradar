#!/usr/bin/env bash
# MedRadar local-state backup (CLOUD-003). Code is recoverable from Git; this covers local state.
set -u
ROOT=/srv/medradar
DEST=$ROOT/var/backup
RETENTION=7
mkdir -p "$DEST"
ts=$(date +%Y%m%d-%H%M%S)
sha=$(sudo -u medradar git -C "$ROOT/app" rev-parse HEAD 2>/dev/null || echo unknown)

tar -czf "$DEST/medradar-state-$ts.tar.gz" -C "$ROOT" data/identity data/cache var/DEPLOYED_SHA 2>/dev/null || true

{
  echo "backup_ts=$ts"
  echo "deployed_sha=$sha"
  echo "host=$(hostname)"
  echo "entries:"
  tar -tzf "$DEST/medradar-state-$ts.tar.gz" 2>/dev/null | sed 's/^/  /'
} > "$DEST/manifest-$ts.txt"

ls -1t "$DEST"/medradar-state-*.tar.gz 2>/dev/null | tail -n +$((RETENTION+1)) | while read -r f; do rm -f "$f"; done
ls -1t "$DEST"/manifest-*.txt 2>/dev/null | tail -n +$((RETENTION+1)) | while read -r f; do rm -f "$f"; done

echo "backup ok: $DEST/medradar-state-$ts.tar.gz sha=$sha"
